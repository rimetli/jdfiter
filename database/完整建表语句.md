# AI 招聘简历筛选系统：完整 MySQL 建表语句

本文档是当前项目数据库结构的 MySQL 8 建表快照，便于部署人员、审计人员或其他开发者在不阅读 Python 代码的情况下查看完整表结构。

- 对应 Alembic 修订：`a91d3e7f5b62`
- 字符集：`utf8mb4`
- 数据库名：`resume_screening`（可按需替换）
- 权威来源仍为 `backend/migrations/versions/`；后续新增迁移时，请同步更新本文档。

## 使用方式

仅适用于**全新的空数据库**。将下面 SQL 代码块保存为 `schema.sql` 后执行：

```bash
mysql -h <MYSQL_HOST> -u <管理员账号> -p < schema.sql
```

执行本 SQL 快照后，数据库已经写入当前 Alembic 版本，不需要再执行 `python -m alembic upgrade head`。反之，若使用 Alembic 创建表，则无需再执行本 SQL。

> 本文件不包含业务数据、管理员账户和任何密码；管理员可在系统登录页完成首次初始化。

```sql
CREATE DATABASE IF NOT EXISTS `resume_screening`
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_unicode_ci;

USE `resume_screening`;

CREATE TABLE `organizations` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(200) NOT NULL,
  `status` VARCHAR(32) NOT NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `app_users` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `organization_id` BIGINT NOT NULL,
  `email_ciphertext` TEXT NOT NULL,
  `email_hash` VARCHAR(64) NOT NULL,
  `display_name` VARCHAR(100) NOT NULL,
  `role` VARCHAR(32) NOT NULL,
  `status` VARCHAR(32) NOT NULL,
  `password_hash` VARCHAR(256) NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_app_user_org_email_hash` (`organization_id`, `email_hash`),
  KEY `ix_app_users_organization_id` (`organization_id`),
  CONSTRAINT `fk_app_users_organization_id`
    FOREIGN KEY (`organization_id`) REFERENCES `organizations` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `job_positions` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `organization_id` BIGINT NOT NULL,
  `name` VARCHAR(200) NOT NULL,
  `department` VARCHAR(200) NULL,
  `jd_content` TEXT NOT NULL,
  `status` VARCHAR(32) NOT NULL,
  `active_requirement_version_id` BIGINT NULL,
  `created_by` BIGINT NULL,
  `archived_at` DATETIME NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `ix_job_org_status_updated` (`organization_id`, `status`, `updated_at`),
  CONSTRAINT `fk_job_positions_organization_id`
    FOREIGN KEY (`organization_id`) REFERENCES `organizations` (`id`),
  CONSTRAINT `fk_job_positions_created_by`
    FOREIGN KEY (`created_by`) REFERENCES `app_users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `job_requirement_versions` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `job_id` BIGINT NOT NULL,
  `version_no` INT NOT NULL,
  `summary` TEXT NULL,
  `rubric_version` VARCHAR(50) NOT NULL,
  `weight_config` JSON NOT NULL,
  `status` VARCHAR(32) NOT NULL,
  `created_by` BIGINT NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `published_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_requirement_version_job_no` (`job_id`, `version_no`),
  KEY `ix_job_requirement_versions_job_id` (`job_id`),
  CONSTRAINT `fk_job_requirement_versions_job_id`
    FOREIGN KEY (`job_id`) REFERENCES `job_positions` (`id`),
  CONSTRAINT `fk_job_requirement_versions_created_by`
    FOREIGN KEY (`created_by`) REFERENCES `app_users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `requirement_items` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `requirement_version_id` BIGINT NOT NULL,
  `dimension_code` VARCHAR(50) NOT NULL,
  `item_code` VARCHAR(100) NOT NULL,
  `name` VARCHAR(200) NOT NULL,
  `description` TEXT NULL,
  `requirement_type` VARCHAR(32) NOT NULL,
  `max_score` NUMERIC(5,2) NOT NULL,
  `is_gate` BOOLEAN NOT NULL,
  `acceptable_alternatives` JSON NULL,
  `evidence_rule` TEXT NULL,
  `sort_order` INT NOT NULL,
  PRIMARY KEY (`id`),
  KEY `ix_requirement_items_requirement_version_id` (`requirement_version_id`),
  CONSTRAINT `fk_requirement_items_requirement_version_id`
    FOREIGN KEY (`requirement_version_id`) REFERENCES `job_requirement_versions` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `candidates` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `organization_id` BIGINT NOT NULL,
  `owner_user_id` BIGINT NULL,
  `name_ciphertext` TEXT NULL,
  `name_hash` VARCHAR(64) NULL,
  `email_ciphertext` TEXT NULL,
  `email_hash` VARCHAR(64) NULL,
  `phone_ciphertext` TEXT NULL,
  `phone_hash` VARCHAR(64) NULL,
  `source` VARCHAR(100) NULL,
  `consent_status` VARCHAR(32) NOT NULL,
  `retention_until` DATETIME NULL,
  `deleted_at` DATETIME NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `ix_candidate_org_email_hash` (`organization_id`, `email_hash`),
  KEY `ix_candidate_org_phone_hash` (`organization_id`, `phone_hash`),
  KEY `ix_candidates_owner_user_id` (`owner_user_id`),
  CONSTRAINT `fk_candidates_organization_id`
    FOREIGN KEY (`organization_id`) REFERENCES `organizations` (`id`),
  CONSTRAINT `fk_candidates_owner_user_id`
    FOREIGN KEY (`owner_user_id`) REFERENCES `app_users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `candidate_identity_claims` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `organization_id` BIGINT NOT NULL,
  `owner_user_id` BIGINT NULL,
  `identity_type` VARCHAR(16) NOT NULL,
  `identity_hash` VARCHAR(64) NOT NULL,
  `candidate_id` BIGINT NOT NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_candidate_identity_claim` (`organization_id`, `owner_user_id`, `identity_type`, `identity_hash`),
  KEY `ix_candidate_identity_claim_organization` (`organization_id`),
  KEY `ix_candidate_claims_owner_user_id` (`owner_user_id`),
  KEY `ix_candidate_identity_claim_candidate` (`candidate_id`),
  CONSTRAINT `fk_candidate_identity_claims_organization_id`
    FOREIGN KEY (`organization_id`) REFERENCES `organizations` (`id`),
  CONSTRAINT `fk_candidate_identity_claims_owner_user_id`
    FOREIGN KEY (`owner_user_id`) REFERENCES `app_users` (`id`),
  CONSTRAINT `fk_candidate_identity_claims_candidate_id`
    FOREIGN KEY (`candidate_id`) REFERENCES `candidates` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `resume_files` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `candidate_id` BIGINT NOT NULL,
  `storage_key` VARCHAR(500) NOT NULL,
  `original_filename` VARCHAR(255) NOT NULL,
  `mime_type` VARCHAR(100) NOT NULL,
  `size_bytes` BIGINT NOT NULL,
  `sha256` VARCHAR(64) NOT NULL,
  `malware_status` VARCHAR(32) NOT NULL,
  `preparsed_text_storage_key` VARCHAR(500) NULL,
  `uploaded_by` BIGINT NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `deleted_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  KEY `ix_resume_files_candidate_id` (`candidate_id`),
  KEY `ix_resume_files_sha256` (`sha256`),
  CONSTRAINT `fk_resume_files_candidate_id`
    FOREIGN KEY (`candidate_id`) REFERENCES `candidates` (`id`),
  CONSTRAINT `fk_resume_files_uploaded_by`
    FOREIGN KEY (`uploaded_by`) REFERENCES `app_users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `resume_parse_versions` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `resume_file_id` BIGINT NOT NULL,
  `version_no` INT NOT NULL,
  `status` VARCHAR(32) NOT NULL,
  `parser_version` VARCHAR(50) NOT NULL,
  `ocr_used` BOOLEAN NOT NULL,
  `normalized_text_storage_key` VARCHAR(500) NULL,
  `profile_json` JSON NULL,
  `validation_errors` JSON NULL,
  `started_at` DATETIME NULL,
  `completed_at` DATETIME NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_resume_parse_version_file_no` (`resume_file_id`, `version_no`),
  CONSTRAINT `fk_resume_parse_versions_resume_file_id`
    FOREIGN KEY (`resume_file_id`) REFERENCES `resume_files` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `job_applications` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `job_id` BIGINT NOT NULL,
  `candidate_id` BIGINT NOT NULL,
  `resume_file_id` BIGINT NOT NULL,
  `stage` VARCHAR(32) NOT NULL,
  `source` VARCHAR(100) NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `ix_application_job_stage_created` (`job_id`, `stage`, `created_at`),
  CONSTRAINT `fk_job_applications_job_id`
    FOREIGN KEY (`job_id`) REFERENCES `job_positions` (`id`),
  CONSTRAINT `fk_job_applications_candidate_id`

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `processing_tasks` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `organization_id` BIGINT NOT NULL,
  `task_type` VARCHAR(50) NOT NULL,

  `status` VARCHAR(32) NOT NULL,
  `progress` INT NOT NULL,
  `attempt_count` INT NOT NULL,
  `priority` INT NOT NULL,
  `available_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `locked_by` VARCHAR(100) NULL,
  `locked_at` DATETIME NULL,
  `error_code` VARCHAR(100) NULL,
  `error_message_safe` TEXT NULL,
  `input_hash` VARCHAR(64) NOT NULL,
  `started_at` DATETIME NULL,
  `completed_at` DATETIME NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `ix_processing_tasks_input_hash` (`input_hash`),
  KEY `ix_task_claim` (`status`, `available_at`, `priority`),
  KEY `ix_task_org_created` (`organization_id`, `created_at`),
  CONSTRAINT `fk_processing_tasks_organization_id`
    FOREIGN KEY (`organization_id`) REFERENCES `organizations` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `candidate_evaluations` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `application_id` BIGINT NOT NULL,
  `parse_version_id` BIGINT NOT NULL,
  `requirement_version_id` BIGINT NOT NULL,
  `status` VARCHAR(32) NOT NULL,
  `total_score` NUMERIC(5,2) NULL,
  `level` VARCHAR(32) NULL,
  `gate_result` VARCHAR(32) NULL,
  `summary_json` JSON NULL,
  `confidence` NUMERIC(4,3) NULL,
  `rubric_version` VARCHAR(50) NOT NULL,
  `prompt_version` VARCHAR(50) NOT NULL,
  `model_provider` VARCHAR(50) NOT NULL,
  `model_name` VARCHAR(100) NOT NULL,
  `supersedes_evaluation_id` BIGINT NULL,
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

  PRIMARY KEY (`id`),
  KEY `ix_evaluation_application_created` (`application_id`, `created_at`),
  CONSTRAINT `fk_candidate_evaluations_application_id`
    FOREIGN KEY (`application_id`) REFERENCES `job_applications` (`id`),

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `evaluation_score_details` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,
  `evaluation_id` BIGINT NOT NULL,

CREATE TABLE `evaluation_evidences` (
  `id` BIGINT NOT NULL AUTO_INCREMENT,

CREATE TABLE `human_decisions` (

CREATE TABLE `alembic_version` (
```

## 说明

`updated_at` 由应用层 SQLAlchemy 在更新记录时维护；本快照与现有迁移保持一致，没有添加 MySQL 的 `ON UPDATE CURRENT_TIMESTAMP` 隐式行为。涉及候选人姓名、邮箱、电话的数据在库中以密文与哈希形式保存，SQL 不会插入明文个人信息。
